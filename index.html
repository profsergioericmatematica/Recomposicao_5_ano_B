// ========================================================== // ANIMAÇÃO Q13 - Divisão Perfeita (3045 ÷ 3) Igual ao Caderno // ========================================================== let passoDivisao = 0; let intervaloDivisao = null; function animarDivisaoQ13() { const canvas = document.getElementById('canvasQ13'); const ctx = canvas.getContext('2d'); const info = document.getElementById('infoQ13'); // Limpa e configura o canvas ctx.clearRect(0, 0, canvas.width, canvas.height); ctx.font = 'bold 28px Arial'; ctx.textAlign = 'center'; ctx.textBaseline = 'middle'; // Configuração de layout (como um caderno) const xInicio = 80; // Coluna inicial do primeiro dígito do dividendo (3) const gapDigito = 30; // Espaço entre os dígitos do dividendo const yTopo = 70; // Linha de cima (onde fica o 3045) const yQuociente = 50; // Linha onde fica a resposta (1 0 1 5) // Posições X de cada dígito do dividendo const x1 = xInicio; const x2 = xInicio + gapDigito; const x3 = xInicio + 2 * gapDigito; const x4 = xInicio + 3 * gapDigito; // Posição do quociente (afastado para a direita, como no papel) const xQ1 = x4 + 80; const xQ2 = xQ1 + gapDigito; const xQ3 = xQ2 + gapDigito; const xQ4 = xQ3 + gapDigito; // 1. Desenhar a parte fixa (Dividendo, Divisor e Chave) ctx.fillStyle = '#333'; // Divisor (3) à esquerda ctx.fillText('3', xInicio + 150, yTopo); // Dividendo (3 0 4 5) no topo ctx.fillText('3', x1, yTopo); ctx.fillText('0', x2, yTopo); ctx.fillText('4', x3, yTopo); ctx.fillText('5', x4, yTopo); // Chave da Divisão (Linha em L que envolve o dividendo) ctx.beginPath(); ctx.moveTo(xInicio + 130, yTopo - 20); // Topo esquerdo (ao lado do 3) ctx.lineTo(xInicio + 130, yTopo + 30); // Desce a linha vertical ctx.lineTo(xQ4 + 20, yTopo + 30); // Vai para a direita embaixo ctx.stroke(); // Zera passos e inicia passoDivisao = 0; info.textContent = "Preparando a conta armada..."; function executarPasso() { passoDivisao++; ctx.font = 'bold 32px Arial'; if (passoDivisao === 1) { // Passo 1: 3 ÷ 3 = 1 ctx.fillStyle = '#d32f2f'; ctx.fillText('1', xQ1, yQuociente+70); // Quociente // -3 (multiplicação) ctx.fillStyle = '#2e7d32'; ctx.fillText('-3', x1, yTopo + 40); // Linha da subtração ctx.beginPath(); ctx.moveTo(x1 - 20, yTopo + 52); ctx.lineTo(x1 + 20, yTopo + 52); ctx.stroke(); // Resultado 0 ctx.fillText('0', x1, yTopo + 75); // Desce o 0 ctx.fillStyle = '#333'; ctx.fillText('0', x2, yTopo + 75); info.textContent = "1º: 3 ÷ 3 = 1 (3 - 3 = 0)"; } else if (passoDivisao === 2) { // Passo 2: 0 ÷ 3 = 0 ctx.fillStyle = '#d32f2f'; ctx.fillText('0', xQ2, yQuociente+70); ctx.fillStyle = '#2e7d32'; ctx.fillText('-0', x2, yTopo + 105); ctx.beginPath(); ctx.moveTo(x2 - 20, yTopo + 117); ctx.lineTo(x2 + 20, yTopo + 117); ctx.stroke(); ctx.fillText('0', x2, yTopo + 140); // Desce o 4 ctx.fillStyle = '#333'; ctx.fillText('4', x3, yTopo + 140); info.textContent = "2º: 0 ÷ 3 = 0 (0 - 0 = 0)"; } else if (passoDivisao === 3) { // Passo 3: 4 ÷ 3 = 1 (resta 1) ctx.fillStyle = '#d32f2f'; ctx.fillText('1', xQ3, yQuociente+70); ctx.fillStyle = '#2e7d32'; ctx.fillText('-3', x3, yTopo + 170); ctx.beginPath(); ctx.moveTo(x3 - 20, yTopo + 182); ctx.lineTo(x3 + 20, yTopo + 182); ctx.stroke(); ctx.fillText('1', x3, yTopo + 205); // Desce o 5 (formando 15) ctx.fillStyle = '#333'; ctx.fillText('5', x4, yTopo + 205); info.textContent = "3º: 4 ÷ 3 = 1 (4 - 3 = 1)"; } else if (passoDivisao === 4) { // Passo 4: 15 ÷ 3 = 5 ctx.fillStyle = '#d32f2f'; ctx.fillText('5', xQ4, yQuociente+70); ctx.fillStyle = '#2e7d32'; ctx.fillText('-1 5', x4 - 30, yTopo + 235); ctx.beginPath(); ctx.moveTo(x4 - 40, yTopo + 247); ctx.lineTo(x4 + 20, yTopo + 247); ctx.stroke(); ctx.fillText('0', x4 - 10, yTopo + 270); info.textContent = "4º: 15 ÷ 3 = 5 (15 - 15 = 0)"; } else if (passoDivisao === 5) { // Resultado final ctx.fillStyle = '#d32f2f'; ctx.font = 'bold 36px Arial'; info.textContent = "Resultado: 3045 ÷ 3 = 1015"; } else { clearInterval(intervaloDivisao); info.textContent = "Fim da conta! 3045 ÷ 3 = 1015"; } } intervaloDivisao = setInterval(executarPasso, 1500); } function resetarDivisaoQ13() { if (intervaloDivisao) { clearInterval(intervaloDivisao); intervaloDivisao = null; } const canvas = document.getElementById('canvasQ13'); const ctx = canvas.getContext('2d'); ctx.clearRect(0, 0, canvas.width, canvas.height); document.getElementById('infoQ13').textContent = "Clique em 'Resolver Passo a Passo' para ver a conta armada."; }
